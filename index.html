Database Schema & Supabase Setup

Run this SQL script in your Supabase SQL Editor to establish all necessary tables, triggers, and RPC verification functions.

-- ==============================================================================
-- SBIMS OFFICIAL CERTIFICATE MANAGEMENT & VERIFICATION SCHEMA MIGRATION
-- Target Database: Supabase PostgreSQL
-- ==============================================================================

-- 1. Create Admin Users Allowlist Table
CREATE TABLE IF NOT EXISTS public.admin_users (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    email TEXT UNIQUE NOT NULL,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- Seed initial default admin users
INSERT INTO public.admin_users (email) 
VALUES ('admin@sbims.edu.in'), ('director@sbims.edu.in')
ON CONFLICT (email) DO NOTHING;

-- 2. Create Certificates Sequence for Yearly Numbering
CREATE SEQUENCE IF NOT EXISTS cert_number_seq START WITH 1 INCREMENT BY 1;

-- 3. Create Certificates Table
CREATE TABLE IF NOT EXISTS public.certificates (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    certificate_id TEXT UNIQUE NOT NULL,
    verification_code TEXT NOT NULL DEFAULT encode(gen_random_bytes(6), 'hex'),
    recipient_name TEXT NOT NULL,
    designation TEXT NOT NULL,
    department TEXT NOT NULL,
    certificate_type TEXT NOT NULL,
    issue_date DATE NOT NULL DEFAULT CURRENT_DATE,
    approving_authority TEXT NOT NULL,
    recognition_note TEXT,
    status TEXT NOT NULL DEFAULT 'VALID' CHECK (status IN ('VALID', 'CANCELLED')),
    cancelled_at TIMESTAMP WITH TIME ZONE,
    cancelled_by TEXT,
    cancellation_reason TEXT,
    created_by TEXT NOT NULL DEFAULT 'system',
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 4. Create Activity Logs Table
CREATE TABLE IF NOT EXISTS public.activity_logs (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    admin_email TEXT NOT NULL,
    action TEXT NOT NULL,
    certificate_id TEXT,
    details TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 5. Automatic Certificate ID Generator Function (SBIMSYYYY00001)
CREATE OR REPLACE FUNCTION public.make_certificate_id()
RETURNS TEXT AS $$
DECLARE
    current_yr TEXT;
    seq_val INT;
    new_cert_id TEXT;
BEGIN
    current_yr := TO_CHAR(CURRENT_DATE, 'YYYY');
    SELECT nextval('cert_number_seq') INTO seq_val;
    new_cert_id := 'SBIMS' || current_yr || LPAD(seq_val::TEXT, 5, '0');
    RETURN new_cert_id;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- 6. Trigger Before Insert to Auto-populate Certificate ID if missing
CREATE OR REPLACE FUNCTION public.set_certificate_id_trigger()
RETURNS TRIGGER AS $$
BEGIN
    IF NEW.certificate_id IS NULL OR NEW.certificate_id = '' THEN
        NEW.certificate_id := public.make_certificate_id();
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS certificates_generate_id ON public.certificates;
CREATE TRIGGER certificates_generate_id
BEFORE INSERT ON public.certificates
FOR EACH ROW EXECUTE FUNCTION public.set_certificate_id_trigger();

-- 7. Public Secure Verification RPC Function
CREATE OR REPLACE FUNCTION public.verify_certificate_public(cert_num TEXT)
RETURNS TABLE (
    certificate_id TEXT,
    recipient_name TEXT,
    designation TEXT,
    department TEXT,
    certificate_type TEXT,
    issue_date DATE,
    approving_authority TEXT,
    recognition_note TEXT,
    status TEXT,
    cancelled_at TIMESTAMP WITH TIME ZONE,
    cancellation_reason TEXT
) AS $$
BEGIN
    RETURN QUERY
    SELECT 
        c.certificate_id,
        c.recipient_name,
        c.designation,
        c.department,
        c.certificate_type,
        c.issue_date,
        c.approving_authority,
        c.recognition_note,
        c.status,
        c.cancelled_at,
        c.cancellation_reason
    FROM public.certificates c
    WHERE UPPER(c.certificate_id) = UPPER(TRIM(cert_num));
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- 8. Enable Row Level Security (RLS)
ALTER TABLE public.certificates ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.admin_users ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.activity_logs ENABLE ROW LEVEL SECURITY;

-- Allow Public Read Access for Verification RPC Function
CREATE POLICY "Public Read Verified Fields" ON public.certificates
    FOR SELECT USING (true);

-- Allow Authenticated Admin Write Operations
CREATE POLICY "Admin Full Access Certificates" ON public.certificates
    FOR ALL USING (auth.role() = 'authenticated');

CREATE POLICY "Admin Full Access Logs" ON public.activity_logs
    FOR ALL USING (auth.role() = 'authenticated');
